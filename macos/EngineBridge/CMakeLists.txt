cmake_minimum_required(VERSION 3.25)
project(GogakuEngine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED GOGAKU_DEPS)
  message(FATAL_ERROR "Pass -DGOGAKU_DEPS with pinned dependency cache")
endif()
find_path(BOOST_INCLUDE boost/version.hpp PATHS /usr/local/opt/boost/include /opt/homebrew/opt/boost/include REQUIRED)
add_library(GogakuEngine SHARED engine.cpp)
target_include_directories(GogakuEngine PRIVATE "${GOGAKU_DEPS}/source/src" "${GOGAKU_DEPS}/build/src" "${GOGAKU_DEPS}/prefix/include/Fcitx5/Utils" "${BOOST_INCLUDE}" PUBLIC include)
target_link_directories(GogakuEngine PRIVATE "${GOGAKU_DEPS}/build/bin" "${GOGAKU_DEPS}/prefix/lib")
target_link_libraries(GogakuEngine PRIVATE IMEPinyin IMECore Fcitx5Utils)
set_target_properties(GogakuEngine PROPERTIES BUILD_RPATH "${GOGAKU_DEPS}/build/bin;${GOGAKU_DEPS}/prefix/lib" INSTALL_NAME_DIR "@rpath")
